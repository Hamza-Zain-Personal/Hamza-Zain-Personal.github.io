# Required libraries import requests import pandas as pd import plotly.graph_objects as go from datetime import datetime import numpy as np import time # API details api_key = "YOUR_API_KEY" # Replace with your actual API key base_url = "https://min-api.cryptocompare.com/data/v2/histoday" # Cryptocurrency list crypto_list = { "BTC": "Bitcoin", "ETH": "Ethereum", "ADA": "Cardano", "SOL": "Solana", "MATIC": "Polygon", "LINK": "Chainlink", "AVAX": "Avalanche", "USDT": "Tether", "USDC": "USD Coin", "BNB": "Binance Coin", "TON": "Toncoin", # Added TON "ARB": "Arbitrum", # Added ARB "RATS": "Rats", # Added RATS "PEPE": "Pepe", # Added PEPE "Ladys": "Milady Meme Coin", # Added Ladys "REPEL": "Repel", # Added REPEL "SUI": "Sui", # Added SUI "Laika": "Laika", # Added Laika "Pengu": "Pengu", # Added Pengu "GT": "GateToken", # Added GT } # Function to fetch historical data with retries and timeout def fetch_price_history(crypto, days, retries=3, delay=5): url = f"{base_url}?fsym={crypto}&tsym=USD&limit={days}&api_key={api_key}" for attempt in range(retries): try: response = requests.get(url, timeout=10) # Set a timeout of 10 seconds response.raise_for_status() # Raise an exception for HTTP errors data = response.json() if data.get("Response") == "Success": return data["Data"]["Data"] else: print(f"Warning: No historical data found for {crypto}") return None except (requests.exceptions.RequestException, ValueError) as e: print(f"Attempt {attempt + 1} failed for {crypto}: {e}") if attempt < retries - 1: time.sleep(delay) # Wait before retrying else: print(f"Max retries reached for {crypto}. Skipping...") return None # Function to calculate EMA (Exponential Moving Average) def calculate_ema(data, period): return data["close"].ewm(span=period, adjust=False).mean() # Function to calculate RSI (Relative Strength Index) def calculate_rsi(data, period=14): delta = data["close"].diff() gain = (delta.where(delta > 0, 0)).rolling(window=period).mean() loss = (-delta.where(delta < 0, 0)).rolling(window=period).mean() rs = gain / loss rsi = 100 - (100 / (1 + rs)) return rsi # Function to calculate support and resistance levels def calculate_support_resistance(data): support = data["low"].rolling(window=14).min().iloc[-1] resistance = data["high"].rolling(window=14).max().iloc[-1] return support, resistance # Map duration options to days duration_options = { "1 week": 7, "2 weeks": 14, "1 month": 30, "6 months": 180, "1 year": 365 } # Prompt user for duration print("Select the price history duration:") for i, option in enumerate(duration_options.keys(), start=1): print(f"{i}. {option}") choice = int(input("Enter your choice (1-5): ")) if choice < 1 or choice > 5: print("Invalid choice. Exiting.") exit() # Get the selected duration selected_duration = list(duration_options.keys())[choice - 1] days = duration_options[selected_duration] print(f"\nFetching historical data for the past {selected_duration}...\n") # Fetch data and filter out coins with no historical data valid_data = [] for crypto, name in crypto_list.items(): history_data = fetch_price_history(crypto, days) if history_data: df = pd.DataFrame(history_data) df["time"] = pd.to_datetime(df["time"], unit="s") # Convert timestamp to datetime # Calculate technical indicators df["EMA_50"] = calculate_ema(df, 50) df["EMA_200"] = calculate_ema(df, 200) df["RSI"] = calculate_rsi(df) support, resistance = calculate_support_resistance(df) valid_data.append((crypto, name, df, support, resistance)) print(f"Processed data for {name} ({crypto}):") print(df[["time", "close", "EMA_50", "EMA_200", "RSI"]].tail()) # Debugging: Print the last few rows # Check if there's any valid data if not valid_data: print("No valid historical data found for any cryptocurrency. Exiting.") exit() # Create a list to store all figures figures = [] # Generate a separate chart for each cryptocurrency for crypto, name, df, support, resistance in valid_data: # Create a new figure fig = go.Figure() # Add traces for price, EMAs, and support/resistance levels fig.add_trace( go.Scatter(x=df["time"], y=df["close"], mode="lines", name="Close Price", line=dict(color="black", width=2)) ) fig.add_trace( go.Scatter(x=df["time"], y=df["EMA_50"], mode="lines", name="50-day EMA", line=dict(color="yellow", width=1.5, dash="dash")) ) fig.add_trace( go.Scatter(x=df["time"], y=df["EMA_200"], mode="lines", name="200-day EMA", line=dict(color="red", width=1.5, dash="dash")) ) fig.add_trace( go.Scatter(x=df["time"], y=[support] * len(df), mode="lines", name="Support (coin tends to stop falling and move up)", line=dict(color="limegreen", width=1.5, dash="dot")) ) fig.add_trace( go.Scatter(x=df["time"], y=[resistance] * len(df), mode="lines", name="Resistance (coin tends to stop rising and move down)", line=dict(color="orange", width=1.5, dash="dot")) ) # Add RSI as a secondary y-axis fig.add_trace( go.Scatter(x=df["time"], y=df["RSI"], mode="lines", name="RSI (above 70 = Sell / below 30 = Buy)", line=dict(color="blue", width=1.5), yaxis="y2") ) # Update layout for the chart fig.update_layout( title=f"{name} ({crypto}) Price History with Technical Indicators ({selected_duration})", xaxis=dict(title="Date", tickangle=45), yaxis=dict(title="Price (USD)"), yaxis2=dict(title="RSI", overlaying="y", side="right", range=[0, 100]), showlegend=True, # Show legend to specify what each line represents template="plotly_white", # Use a clean theme margin=dict(l=50, r=50, b=150, t=100, pad=10), # Adjust margins to accommodate annotations font=dict(size=12) # Set font size ) # Add horizontal lines for RSI thresholds fig.add_hline(y=30, line=dict(color="red", dash="dash", width=1), yref="y2") fig.add_hline(y=70, line=dict(color="green", dash="dash", width=1), yref="y2") # Check for Golden Cross (50-day EMA > 200-day EMA) if df["EMA_50"].iloc[-1] > df["EMA_200"].iloc[-1]: fig.add_annotation( x=0.5, y=-0.2, xref="paper", yref="paper", text="Golden Cross = A potential buy", showarrow=False, font=dict(size=12, color="green"), xanchor="center", yanchor="top" ) # Check for RSI conditions rsi_value = df["RSI"].iloc[-1] if rsi_value <= 30: fig.add_annotation( x=0.5, y=-0.3, xref="paper", yref="paper", text="Buying opportunity based on RSI (oversold)", showarrow=False, font=dict(size=12, color="blue"), xanchor="center", yanchor="top" ) elif rsi_value > 70: fig.add_annotation( x=0.5, y=-0.3, xref="paper", yref="paper", text="Selling opportunity based on RSI (overbought)", showarrow=False, font=dict(size=12, color="red"), xanchor="center", yanchor="top" ) # Add the figure to the list figures.append(fig) # Save all figures to a single HTML file with open("crypto_charts.html", "w") as f: for fig in figures: f.write(fig.to_html(full_html=False, include_plotlyjs="cdn")) print("All charts saved to crypto_charts.html. Open the file in your browser to view them.")
